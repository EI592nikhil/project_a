document.addEventListener("click", function (e) {
    const sup = e.target.closest(".cdp-hero sup.cmp-superscript__trigger");

    if (!sup) return;

    const cdpHero = sup.closest(".cdp-hero");

    let heroTitle = "";
    const heroLayer = cdpHero?.getAttribute("data-cmp-data-layer");

    if (heroLayer) {
        try {
            const parsed = JSON.parse(heroLayer);
            const key = Object.keys(parsed)[0];
            heroTitle = parsed[key]?.title || "";
        } catch (err) {}
    }

    if (!heroTitle) {
        heroTitle =
            cdpHero?.querySelector(".cdp-hero__title")?.textContent.trim() || "";
    }

    adobeDataLayer.push({
        event: adobeDataLayerEvents.component.linkClick,
        linkDetails: {
            clickName: "sup",
            clickDestination: "NA",
            clickLocation: "body",
            clickType: "other",
            clickContext: heroTitle.toLowerCase(),
            componentContext: "content detail hero"
        }
    });
});
