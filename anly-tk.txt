describe("CDP Hero Share Button", () => {
  beforeEach(() => setupAdobeDataLayer());

  test("should push analytics for share button using data layer title", () => {
    document.body.innerHTML = `
      <div class="cdp-hero"
           data-cmp-data-layer='{"hero":{"title":"Hero Share"}}'>
        <button class="cdp-hero__share-btn">Share</button>
      </div>
    `;

    loadAnalyticsModule();

    document.querySelector(".cdp-hero__share-btn").click();

    expect(lastPush().linkDetails).toEqual(
      expect.objectContaining({
        clickName: "share",
        clickDestination: "NA",
        clickType: "other",
        clickContext: "hero share",
        componentContext: "content detail hero"
      })
    );
  });

  test("should fall back to hero title for share button when data layer is invalid", () => {
    document.body.innerHTML = `
      <div class="cdp-hero"
           data-cmp-data-layer="invalid-json">
        <h1 class="cdp-hero__title">Fallback Share Hero</h1>
        <button class="cdp-hero__share-btn">Share</button>
      </div>
    `;

    loadAnalyticsModule();

    document.querySelector(".cdp-hero__share-btn").click();

    expect(lastPush().linkDetails.clickContext).toBe("fallback share hero");
    expect(lastPush().linkDetails.clickDestination).toBe("NA");
  });
});

describe("CDP Hero Anchor Links", () => {
  beforeEach(() => setupAdobeDataLayer());

  test("should push analytics for hero anchor link", () => {
    document.body.innerHTML = `
      <div class="cdp-hero"
           data-cmp-data-layer='{"hero":{"title":"Anchor Hero"}}'>
        <button
          class="cdp-hero__anchor-link"
          data-target="overview">
          Overview
        </button>
      </div>
    `;

    loadAnalyticsModule();

    document.querySelector(".cdp-hero__anchor-link").click();

    expect(lastPush().linkDetails.clickName).toBe("overview");
    expect(lastPush().linkDetails.clickContext).toBe("anchor hero");
    expect(lastPush().linkDetails.componentContext).toBe("content detail hero");
    expect(lastPush().linkDetails.clickDestination).toContain("#overview");
  });


  test("should use hero title fallback for anchor link when data layer is missing", () => {
    document.body.innerHTML = `
      <div class="cdp-hero">
        <h1 class="cdp-hero__title">Anchor Fallback</h1>
        <button
          class="cdp-hero__anchor-link"
          data-target="features">
          Features
        </button>
      </div>
    `;

    loadAnalyticsModule();

    document.querySelector(".cdp-hero__anchor-link").click();

    expect(lastPush().linkDetails.clickContext).toBe("anchor fallback");
    expect(lastPush().linkDetails.clickDestination).toContain("#features");
  });


  test("should set clickDestination as NA when data-target is missing", () => {
    document.body.innerHTML = `
      <div class="cdp-hero">
        <h1 class="cdp-hero__title">Anchor Hero</h1>
        <button class="cdp-hero__anchor-link">
          Overview
        </button>
      </div>
    `;

    loadAnalyticsModule();

    document.querySelector(".cdp-hero__anchor-link").click();

    expect(lastPush().linkDetails.clickDestination).toBe("NA");
  });
});

// -----------------------------
// CDP Hero - Sup Tag Click
// -----------------------------
const supTags = document.querySelectorAll(".cdp-hero sup");

supTags.forEach((sup) => {
    sup.addEventListener("click", function () {

        const cdpHero = sup.closest(".cdp-hero");
        let heroTitle = "";
        const heroLayer = cdpHero?.getAttribute("data-cmp-data-layer");

        if (heroLayer) {
            try {
                const parsed = JSON.parse(heroLayer);
                const key = Object.keys(parsed)[0];
                heroTitle = parsed[key]?.title || "";
            } catch (e) {}
        }

        if (!heroTitle) {
            heroTitle = cdpHero?.querySelector(".cdp-hero__title")?.textContent.trim() || "";
        }

        adobeDataLayer.push({
            event: adobeDataLayerEvents.component.linkClick,
            linkDetails: {
                clickName: "sup",
                clickDestination: "NA",
                clickLocation: "body",
                clickType: "other",
                clickContext: heroTitle.toLowerCase(),
                componentContext: "content detail hero"
            }
        });

    });
});
