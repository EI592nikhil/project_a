document.addEventListener("DOMContentLoaded", () => {

  /* =====================================
     MOBILE DROPDOWN
  ====================================== */

  const dropdown = document.querySelector(
    ".subheadernav-mobile-dropdown"
  ) as HTMLElement | null;

  const trigger = document.querySelector(
    ".subheadernav-dropdown-trigger"
  ) as HTMLButtonElement | null;

  if (dropdown && trigger) {

    trigger.addEventListener("click", (e: MouseEvent) => {
      e.stopPropagation();
      dropdown.classList.toggle("open");
    });

    document.addEventListener("click", () => {
      dropdown.classList.remove("open");
    });

    dropdown.addEventListener("click", (e: MouseEvent) => {
      e.stopPropagation();
    });
  }

  /* =====================================
     HELPERS
  ====================================== */

  const formatTitle = (value: string) =>
    value
      .replace(/-/g, " ")
      .replace(/\b\w/g, (c) => c.toUpperCase());

  const normalizeClass = (value: string) =>
    value
      .trim()
      .toLowerCase()
      .replace(/\s+/g, "-");

  /* =====================================
     NAV ITEMS
  ====================================== */

  const navItems = document.querySelectorAll<HTMLElement>(
    ".subheadernav-item"
  );

  navItems.forEach((item) => {

    const rawTarget = item.dataset.target;

    if (rawTarget) {
      item.textContent = formatTitle(rawTarget);
    }

    item.addEventListener("click", () => {

      const target = item.dataset.target;
      if (!target) return;

      const normalizedTarget = normalizeClass(target);

      const section = document.querySelector(
        `.${normalizedTarget}`
      ) as HTMLElement | null;

      if (section) {

        const nav = document.querySelector(
          ".pdp-subheadernav"
        ) as HTMLElement | null;

        const navHeight = nav?.offsetHeight || 0;

        const sectionTop =
          section.getBoundingClientRect().top +
          window.pageYOffset -
          navHeight -
          20;

        window.scrollTo({
          top: sectionTop,
          behavior: "smooth"
        });

        navItems.forEach(n =>
          n.classList.remove("active")
        );

        item.classList.add("active");
        dropdown?.classList.remove("open");
      }
    });
  });

  /* =====================================
     FIXED SUBHEADER
  ====================================== */

  const nav = document.querySelector(
    ".pdp-subheadernav"
  ) as HTMLElement | null;

  if (nav) {

    const navTop =
      nav.getBoundingClientRect().top +
      window.scrollY;

    const navHeight = nav.offsetHeight;

    const placeholder = document.createElement("div");

    const onScroll = () => {

      if (window.scrollY >= navTop) {

        if (!nav.classList.contains("fixed-nav")) {

          placeholder.style.height = `${navHeight}px`;
          placeholder.style.width = "100%";

          nav.parentNode?.insertBefore(
            placeholder,
            nav
          );

          nav.classList.add("fixed-nav");
        }

      } else {

        nav.classList.remove("fixed-nav");
        placeholder.remove();
      }
    };

    window.addEventListener("scroll", onScroll);
    window.addEventListener("resize", onScroll);

    onScroll();
  }

  /* =====================================
     SCROLL SPY
  ====================================== */

  const sectionMap: {
    section: HTMLElement;
    id: string;
  }[] = [];

  navItems.forEach((item) => {

    const raw = item.dataset.target;
    if (!raw) return;

    const normalized = normalizeClass(raw);

    const section = document.querySelector(
      `.${normalized}`
    ) as HTMLElement | null;

    if (section) {
      sectionMap.push({
        section,
        id: raw
      });
    }
  });

  const onScrollSpy = () => {

    let active: string | null = null;

    sectionMap.forEach(({ section, id }) => {

      const rect = section.getBoundingClientRect();

      if (rect.top <= 180 && rect.bottom >= 180) {
        active = id;
      }
    });

    if (!active) return;

    navItems.forEach((item) => {

      if (item.dataset.target === active) {
        item.classList.add("active");
      } else {
        item.classList.remove("active");
      }
    });
  };

  window.addEventListener("scroll", onScrollSpy);

  onScrollSpy();
});
