document.addEventListener("DOMContentLoaded", () => {

  /* =====================================
     MOBILE DROPDOWN
  ====================================== */

  const dropdown = document.querySelector(
    ".subheadernav-mobile-dropdown"
  ) as HTMLElement | null;

  const trigger = document.querySelector(
    ".subheadernav-dropdown-trigger"
  ) as HTMLButtonElement | null;

  if (dropdown && trigger) {

    trigger.addEventListener("click", (e: MouseEvent) => {
      e.stopPropagation();
      dropdown.classList.toggle("open");
    });

    document.addEventListener("click", () => {
      dropdown.classList.remove("open");
    });

    dropdown.addEventListener("click", (e: MouseEvent) => {
      e.stopPropagation();
    });
  }

  /* =====================================
     HELPERS
  ====================================== */

  const formatTitle = (value: string) =>
    value
      .replace(/-/g, " ")
      .replace(/\s*&\s*/g, " & ")
      .trim()
      .split(" ")
      .filter(Boolean)
      .map(word =>
        word === "&"
          ? "&"
          : word.charAt(0).toUpperCase() + word.slice(1)
      )
      .join(" ");

  const normalizeClass = (value: string) =>
    value
      .trim()
      .toLowerCase()
      .replace(/&/g, "")
      .replace(/[^a-z0-9\s]/g, "")
      .replace(/\s+/g, "-")
      .replace(/-+/g, "-");

  /* =====================================
     OFFSET FIX
  ====================================== */

  const getOffset = () => {
    const header = document.querySelector(
      ".header.header-v2"
    ) as HTMLElement | null;

    const nav = document.querySelector(
      ".pdp-subheadernav"
    ) as HTMLElement | null;

    return (
      (header?.offsetHeight || 0) +
      (nav?.offsetHeight || 0)
    );
  };

  /* =====================================
     NAV ITEMS
  ====================================== */

  const navItems = document.querySelectorAll<HTMLElement>(
    ".subheadernav-item"
  );

  const selectedText = document.querySelector(
    ".subheadernav-selected"
  ) as HTMLElement | null;


  const updateDropdownSelection = (target: string) => {

    const formattedTitle = formatTitle(target);

    if (selectedText) {
      selectedText.textContent = formattedTitle;
    }

    const mobileItems = document.querySelectorAll<HTMLElement>(
      ".subheadernav-mobile-dropdown .subheadernav-item"
    );

    mobileItems.forEach((item) => {

      item.classList.toggle(
        "active",
        item.dataset.target === target
      );

    });
  };

  navItems.forEach((item) => {

    const rawTarget = item.dataset.target;

    if (rawTarget) {
      item.textContent = formatTitle(rawTarget);
    }

    item.addEventListener("click", () => {

      const target = item.dataset.target;
      if (!target) return;

      const normalizedTarget =
        normalizeClass(target);
      const section = document.querySelector(
        `.${normalizedTarget}`
      ) as HTMLElement | null;

      if (section) {
        const offset = getOffset();
        const sectionTop =
          section.getBoundingClientRect().top +
          window.pageYOffset -
          offset -
          20;
        window.scrollTo({
          top: sectionTop,
          behavior: "smooth"
        });

        navItems.forEach(n =>
          n.classList.remove("active")
        );

        item.classList.add("active");
        updateDropdownSelection(target);

        dropdown?.classList.remove("open");
      }
    });
  });
  /* =====================================
     FIXED SUBHEADER
  ====================================== */

  const nav = document.querySelector(
    ".pdp-subheadernav"
  ) as HTMLElement | null;

  if (nav) {
    const navTop = nav.offsetTop;
    const navHeight = nav.offsetHeight;
    const placeholder = document.createElement("div");
    const header = document.querySelector(
      ".cmp-header-v2-wrapper"
    ) as HTMLElement | null;

    let lastScrollTop = window.scrollY;

    const onScroll = () => {

      const currentScrollTop = window.scrollY;

      if (currentScrollTop >= navTop) {

        if (!nav.classList.contains("fixed-nav")) {

          placeholder.style.height =
            `${navHeight}px`;

          placeholder.style.width = "100%";

          nav.parentNode?.insertBefore(
            placeholder,
            nav
          );

          nav.classList.add("fixed-nav");
        }

        const headerHeight =
          header?.offsetHeight || 0;
        nav.style.top =
          currentScrollTop < lastScrollTop
            ? `${headerHeight}px`
            : "0px";
      } else {
        nav.classList.remove("fixed-nav");
        nav.style.top = "";
        placeholder.remove();
      }
      lastScrollTop = currentScrollTop;

    };

    window.addEventListener("scroll", onScroll);
    window.addEventListener("resize", onScroll);

    onScroll();
  }

  /* =====================================
     SCROLL SPY
  ====================================== */
  const sectionMap: {
    section: HTMLElement;
    id: string;
  }[] = [];

  navItems.forEach((item) => {

    const raw = item.dataset.target;
    if (!raw) return;

    const normalized =
      normalizeClass(raw);
    const sections =
      document.querySelectorAll(
        `.${normalized}`
      );

    sections.forEach((section) => {
      sectionMap.push({
        section: section as HTMLElement,
        id: raw
      });
    });
  });

  const onScrollSpy = () => {

    const lastSection =
      sectionMap[
        sectionMap.length - 1
      ]?.section;

    if (
      lastSection &&
      window.scrollY >
        lastSection.offsetTop +

        lastSection.offsetHeight

    ) {
      navItems.forEach(item =>

        item.classList.remove("active")

      );
      return;

    }

    let active: string | null = null;
    let minDistance = Infinity;

    const viewportCenter =
      window.innerHeight / 2 +
      getOffset() / 2;

    sectionMap.forEach(

      ({ section, id }) => {
        const rect =
          section.getBoundingClientRect();
        const sectionCenter =
          rect.top + rect.height / 2;

        const distance = Math.abs(
          sectionCenter - viewportCenter
        );

        if (distance < minDistance) {
          minDistance = distance;
          active = id;
        }
      }

    );
    if (!active) return;

    navItems.forEach((item) => {
      item.classList.toggle(
        "active",

        item.dataset.target === active
      );
    });

    updateDropdownSelection(active);
  };
  const firstItem = navItems[0];
  if (firstItem?.dataset.target) {
    updateDropdownSelection(
      firstItem.dataset.target
    );
  }

  window.addEventListener(
    "scroll",
    onScrollSpy
  );
  onScrollSpy();
});
